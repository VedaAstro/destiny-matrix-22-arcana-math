B = R22(M)                              // Top Vertex: Spiritual Talent & Intuition
C = R22(sum_digits(Y))                  // Right Vertex: Material Karma & Financial Threshold
D = R22(A + B + C)                      // Bottom Vertex: Past-Cycle Karmic Anchor
E = R22(A + B + C + D)                  // Center Node: Core Comfort Zone of the Soul
KarmicTail = [R22(D + E), R22(D + R22(D + E)), D]
